\section{Installation Guide}
\label{sec:installation_guide}

\subsection{Prerequisites}

\begin{itemize}[leftmargin=*]
    \item Python 3.10 or higher
    \item Node.js 16 or higher (with npm)
    \item A modern web browser (Chrome, Firefox, or Edge)
    \item (Optional) A Google Gemini API key for AI decision support
\end{itemize}

\subsection{Backend Setup}

\begin{lstlisting}[style=codestyle, language=bash, caption=Backend installation and startup]
# Navigate to the backend directory
cd Multi-Agent-System

# Install Python dependencies
pip install -r requirements.txt

# (Optional) Set the Gemini API key
# Create a .env file or set the environment variable:
set GEMINI_API_KEY=your_api_key_here

# (Optional) Generate synthetic input data
python generate_data.py --count 150 --seed 42

# (Optional) Train the ML model
python -m ml.train_severity_model --samples 5000

# Start the FastAPI server
python -m uvicorn app:app --host 0.0.0.0 --port 8000 --reload
\end{lstlisting}

The server starts at \texttt{http://localhost:8000}. The interactive API documentation is available at \texttt{http://localhost:8000/docs}.

\subsection{Frontend Setup}

\begin{lstlisting}[style=codestyle, language=bash, caption=Frontend installation and startup]
# Navigate to the frontend directory
cd "multiagent project"

# Install Node.js dependencies
npm install

# Start the development server
npm start
\end{lstlisting}

The React application starts at \texttt{http://localhost:3000} and connects to the backend at \texttt{http://localhost:8000}.

\subsection{Quick Start (Windows)}

A batch file \texttt{RUN\_PROJECT.bat} is provided at the project root to start both servers simultaneously.

\subsection{Running a Simulation}

\begin{enumerate}[leftmargin=*]
    \item Navigate to the \textbf{Simulation} tab.
    \item Enter a seed value (any integer; use 42 for reproducible results).
    \item Select the strategy: \textbf{Smart} (ML + A* routing) or \textbf{Baseline} (fixed allocation).
    \item Click \textbf{Run Simulation}.
    \item The state tracker updates in real time as the pipeline progresses.
    \item Once complete, metrics appear below the state tracker.
\end{enumerate}

\subsection{Viewing the Map}

\begin{enumerate}[leftmargin=*]
    \item Navigate to the \textbf{Map} tab after a simulation completes.
    \item Incident markers appear at their geographic positions.
    \item Route polylines show the A*-computed paths from teams to incidents.
    \item Use the disaster type and severity dropdowns to filter visible markers.
    \item Toggle hospitals, shelters, and fire stations using the sidebar checkboxes.
\end{enumerate}

\subsection{Admin Panel}

Admin users see an additional \textbf{Admin} tab with:
\begin{itemize}[leftmargin=*]
    \item User management: create new users, change roles, delete users.
    \item System configuration: adjust team counts, capacity, and loss factors.
    \item Rescue team management: add, edit, or view rescue teams.
\end{itemize}

\subsection{ML Management}

The \textbf{ML} tab allows:
\begin{itemize}[leftmargin=*]
    \item Viewing the active model's accuracy, F1 score, and training date.
    \item Triggering model retraining via a button click.
    \item Running one-off severity predictions with custom feature inputs.
\end{itemize}

\section{Database Schema}
\label{sec:appendix_schema}

\begin{lstlisting}[style=codestyle, language=SQL, caption=Complete SQLite schema (condensed)]
CREATE TABLE users (
    user_id INTEGER PRIMARY KEY AUTOINCREMENT,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(120) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role VARCHAR(20) DEFAULT 'operator'
);

CREATE TABLE simulation_runs (
    run_id TEXT PRIMARY KEY,
    user_id INTEGER,
    seed INTEGER NOT NULL,
    strategy VARCHAR(20) DEFAULT 'smart',
    status VARCHAR(20) DEFAULT 'RUNNING',
    started_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    completed_at DATETIME,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
);

CREATE TABLE incidents (
    incident_id INTEGER PRIMARY KEY AUTOINCREMENT,
    run_id TEXT NOT NULL,
    location VARCHAR(120),
    latitude REAL, longitude REAL,
    severity VARCHAR(20),
    people_affected INTEGER,
    disaster_type VARCHAR(50),
    priority VARCHAR(20),
    estimated_risk VARCHAR(20),
    ml_predicted_severity VARCHAR(20),
    ml_confidence REAL,
    assigned_team_id TEXT,
    route_path TEXT,
    route_distance_km REAL,
    route_eta_hours REAL,
    FOREIGN KEY (run_id) REFERENCES simulation_runs(run_id)
);

CREATE TABLE agent_logs (
    log_id INTEGER PRIMARY KEY AUTOINCREMENT,
    run_id TEXT NOT NULL,
    agent_name VARCHAR(60) NOT NULL,
    state VARCHAR(40) NOT NULL,
    output_json TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (run_id) REFERENCES simulation_runs(run_id)
);

CREATE TABLE run_logs (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    run_id TEXT NOT NULL,
    state TEXT NOT NULL,
    payload TEXT,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE run_metrics (
    metric_id INTEGER PRIMARY KEY AUTOINCREMENT,
    run_id TEXT UNIQUE NOT NULL,
    total_incidents INTEGER,
    total_affected INTEGER,
    total_lives_saved INTEGER,
    overall_efficiency VARCHAR(20),
    avg_response_time VARCHAR(30),
    resource_utilization VARCHAR(20),
    status VARCHAR(20),
    FOREIGN KEY (run_id) REFERENCES simulation_runs(run_id)
);

CREATE TABLE system_config (
    key VARCHAR(50) PRIMARY KEY,
    value VARCHAR(255) NOT NULL
);

CREATE TABLE custom_incidents (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    location VARCHAR(120) NOT NULL,
    latitude REAL, longitude REAL,
    severity VARCHAR(20) NOT NULL,
    people_affected INTEGER NOT NULL,
    disaster_type VARCHAR(50) NOT NULL
);

CREATE TABLE agent_messages (
    message_id INTEGER PRIMARY KEY AUTOINCREMENT,
    run_id TEXT NOT NULL,
    sender VARCHAR(60) NOT NULL,
    receiver VARCHAR(60) NOT NULL,
    message_type VARCHAR(30) NOT NULL,
    payload TEXT,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE ml_models (
    model_id INTEGER PRIMARY KEY AUTOINCREMENT,
    model_name VARCHAR(100) NOT NULL,
    model_path VARCHAR(255) NOT NULL,
    accuracy REAL,
    f1_score REAL,
    training_samples INTEGER,
    trained_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    is_active BOOLEAN DEFAULT 1
);

CREATE TABLE rescue_teams (
    team_id VARCHAR(50) PRIMARY KEY,
    team_name VARCHAR(100) NOT NULL,
    latitude REAL NOT NULL,
    longitude REAL NOT NULL,
    status VARCHAR(20) DEFAULT 'available',
    capacity INTEGER DEFAULT 10,
    specialization VARCHAR(50)
);
\end{lstlisting}
